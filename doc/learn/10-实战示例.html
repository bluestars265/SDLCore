10 实战示例

目标：综合运用所学，组装一个可运行的完整界面（开始菜单 + 游戏内 HUD）。

我们将编写一个 demo_scene.py（放在 mods/base_game/，作为主 mod 场景）， 包含两个场景：menu（开始菜单）与 hud（游戏内 HUD）。

1. 开始菜单场景（菜单按钮 + 图标）

# mods/base_game/demo_scene.py
import os
import sdl2
from SDLCore.resource import resources
from SDLCore.scene import Scene
from SDLCore.ui.button import Button
from SDLCore.ui.label import Label
from SDLCore.ui.image import Image
from SDLCore.ui.panel import Panel
from SDLCore.ui.layout import VBoxLayout, HBoxLayout

ROOT = os.path.dirname(os.path.abspath(__file__))
LOGO = os.path.join(ROOT, "resource", "images", "png", "logo.png")


class MenuScene(Scene):
    """开始菜单：logo + 菜单按钮。"""
    def __init__(self, manager):
        super().__init__(manager)
        self._built = False

    def on_enter(self):
        if self._built:
            return
        self._built = True
        self.ui.root_panel.color = (26, 28, 38, 255)

        # logo（保持宽高比居中）
        self.ui.root_panel.add_child(Image((0, 0, -1, 200), path=LOGO))

        # 菜单面板：VBox 垂直排列
        menu = Panel((0, 0, 360, 320), color=(30, 30, 40, 220),
                     layout=VBoxLayout(spacing=12, padding=(30, 20, 30, 20)))
        menu.add_child(Button((0, 0, -1, 52), text="开始游戏",
                              callback=lambda: self.manager.switch("hud")))
        menu.add_child(Button((0, 0, -1, 52), text="游戏设置",
                              callback=lambda: print("打开设置")))
        menu.add_child(Button((0, 0, -1, 52), text="退出",
                              callback=lambda: print("退出游戏")))
        self.ui.root_panel.add_child(menu)
        self._relayout()

    def _relayout(self):
        w, h = self.game.window.size
        # logo 位于上部；菜单面板水平居中
        menu = self.ui.root_panel.children[1]
        mw, mh = 360, 320
        menu.set_rect(((w - mw) // 2, int(h * 0.5), mw, mh))

    def on_resize(self, width, height):
        if getattr(self, "_built", False):
            self._relayout()


class HudScene(Scene):
    """游戏内 HUD：左上角资源面板 + 底部按钮条。"""
    def __init__(self, manager):
        super().__init__(manager)
        self._built = False

    def on_enter(self):
        if self._built:
            return
        self._built = True
        self.ui.root_panel.color = (60, 68, 54, 255)

        # 左上角：资源统计
        panel = Panel((20, 20, 200, 96), color=(0, 0, 0, 160),
                      layout=VBoxLayout(spacing=6, padding=(14, 10, 14, 10)))
        self.hp_label = Label((0, 0, -1, 22), text="生命: 100",
                              color=(255, 255, 255, 255), font_size="16px",
                              factory=self.game.factory)
        self.gold_label = Label((0, 0, -1, 22), text="金币: 50",
                                color=(255, 220, 0, 255), font_size="16px",
                                factory=self.game.factory)
        panel.add_child(self.hp_label)
        panel.add_child(self.gold_label)
        self.ui.root_panel.add_child(panel)

        # 底部：暂停 / 返回菜单按钮
        bar = Panel((0, 0, 320, 60), color=(0, 0, 0, 180),
                    layout=HBoxLayout(spacing=12, padding=(16, 10, 16, 10)))
        bar.add_child(Button((0, 0, 130, -1), text="暂停",
                             callback=lambda: print("暂停游戏")))
        bar.add_child(Button((0, 0, 130, -1), text="返回菜单",
                             callback=lambda: self.manager.switch("menu")))
        self.ui.root_panel.add_child(bar)

    def on_resize(self, width, height):
        if not getattr(self, "_built", False):
            return
        self.ui.root_panel.children[1].set_rect(
            ((width - 320) // 2, height - 80, 320, 60))

    def update(self, delta_time):
        super().update(delta_time)
        # 每帧刷新 HUD 数值（示意）
        if self._built:
            self.hp_label.set_text(f"生命: {100 - int(delta_time * 10)}")


# ---- 注册到 base_game 主 mod（在 main.py 入口中调用） ----

def register_scenes(api):
    """在 base_game 的 register(api) 中调用，注册实战场景。"""
    api.register_scene("menu", MenuScene)
    api.register_scene("hud", HudScene)

2. 注册到 base_game 主 mod

在 mods/base_game/main.py 的 register(api) 中：

demo = _load_mod_module(api, "demo_scene.py")
demo.register_scenes(api)          # 注册 menu / hud 两个场景

3. 运行

python test_main.py                # 或 python main.py

启动后进入 loading → start；通过 SceneManager.switch("menu") 进入开始菜单， 点击"开始游戏"切换到 HUD 场景，HUD 中点击"返回菜单"切回。

4. 复盘：用到了哪些知识
